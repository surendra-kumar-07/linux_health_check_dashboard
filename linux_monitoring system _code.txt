# ==========================================================
# CPU Usage
# ==========================================================

get_cpu_usage() {

    local user1
    local nice1
    local system1
    local idle1
    local iowait1
    local irq1
    local softirq1
    local steal1

    local user2
    local nice2
    local system2
    local idle2
    local iowait2
    local irq2
    local softirq2
    local steal2

    local total1
    local total2
    local idle_total1
    local idle_total2
    local total_diff
    local idle_diff

    read -r cpu user1 nice1 system1 idle1 iowait1 irq1 softirq1 steal1 _ < /proc/stat

    total1=$((user1 + nice1 + system1 + idle1 + iowait1 + irq1 + softirq1 + steal1))
    idle_total1=$((idle1 + iowait1))

    sleep 1

    read -r cpu user2 nice2 system2 idle2 iowait2 irq2 softirq2 steal2 _ < /proc/stat

    total2=$((user2 + nice2 + system2 + idle2 + iowait2 + irq2 + softirq2 + steal2))
    idle_total2=$((idle2 + iowait2))

    total_diff=$((total2 - total1))
    idle_diff=$((idle_total2 - idle_total1))

    if [ "$total_diff" -eq 0 ]; then
        echo "0"
    else
        echo $((100 * (total_diff - idle_diff) / total_diff))
    fi
}

# ==========================================================
# Memory Usage
# ==========================================================

get_memory_usage() {

    local total
    local available
    local used

    total=$(awk '/MemTotal:/ {print $2}' /proc/meminfo)

    available=$(awk '/MemAvailable:/ {print $2}' /proc/meminfo)

    used=$((total - available))

    if [ "$total" -gt 0 ]; then
        echo $((used * 100 / total))
    else
        echo "0"
    fi
}

get_memory_total() {

    free -h | awk '/Mem:/ {print $2}'
}

get_memory_used() {

    free -h | awk '/Mem:/ {print $3}'
}

get_memory_available() {

    free -h | awk '/Mem:/ {print $7}'
}

# ==========================================================
# Disk Usage
# ==========================================================

get_disk_usage() {

    df -P / | awk 'NR==2 {
        gsub("%","",$5);
        print $5
    }'
}

get_disk_total() {

    df -h / | awk 'NR==2 {print $2}'
}

get_disk_used() {

    df -h / | awk 'NR==2 {print $3}'
}

get_disk_free() {

    df -h / | awk 'NR==2 {print $4}'
}

# ==========================================================
# System Information
# ==========================================================

get_load_average() {

    awk '{print $1}' /proc/loadavg
}

get_process_count() {

    find /proc -maxdepth 1 -type d \
        -regex '/proc/[0-9]+' 2>/dev/null |
        wc -l
}

get_cpu_cores() {

    nproc
}

get_uptime() {

    uptime -p
}

get_hostname() {

    hostname
}

get_os() {

    if [ -f /etc/os-release ]; then

        . /etc/os-release

        echo "${PRETTY_NAME:-Unknown Linux}"

    else

        echo "Unknown Linux"

    fi
}

get_kernel() {

    uname -r
}

get_architecture() {

    uname -m
}

# ==========================================================
# Network Interface
# ==========================================================

get_interface() {

    ip route 2>/dev/null |
        awk '/default/ {
            print $5;
            exit
        }'
}

# ==========================================================
# Network RX
# ==========================================================

get_rx() {

    local interface
    local bytes

    interface=$(get_interface)

    if [ -n "$interface" ] &&
       [ -f "/sys/class/net/$interface/statistics/rx_bytes" ]; then

        bytes=$(cat "/sys/class/net/$interface/statistics/rx_bytes")

        if [ "$bytes" -ge 1073741824 ]; then

            awk "BEGIN {
                printf \"%.2f GB\", $bytes/1073741824
            }"

        elif [ "$bytes" -ge 1048576 ]; then

            awk "BEGIN {
                printf \"%.2f MB\", $bytes/1048576
            }"

        elif [ "$bytes" -ge 1024 ]; then

            awk "BEGIN {
                printf \"%.2f KB\", $bytes/1024
            }"

        else

            echo "${bytes} B"

        fi

    else

        echo "N/A"

    fi
}

# ==========================================================
# Network TX
# ==========================================================

get_tx() {

    local interface
    local bytes

    interface=$(get_interface)

    if [ -n "$interface" ] &&
       [ -f "/sys/class/net/$interface/statistics/tx_bytes" ]; then

        bytes=$(cat "/sys/class/net/$interface/statistics/tx_bytes")

        if [ "$bytes" -ge 1073741824 ]; then

            awk "BEGIN {
                printf \"%.2f GB\", $bytes/1073741824
            }"

        elif [ "$bytes" -ge 1048576 ]; then

            awk "BEGIN {
                printf \"%.2f MB\", $bytes/1048576
            }"

        elif [ "$bytes" -ge 1024 ]; then

            awk "BEGIN {
                printf \"%.2f KB\", $bytes/1024
            }"

        else

            echo "${bytes} B"

        fi

    else

        echo "N/A"

    fi
}

# ==========================================================
# Service Status
# ==========================================================

service_status() {

    local service="$1"

    if systemctl list-unit-files \
        "${service}.service" 2>/dev/null |
        grep -q "^${service}\.service"; then

        if systemctl is-active --quiet "$service"; then

            echo "RUNNING"

        else

            echo "STOPPED"

        fi

    else

        echo "NOT INSTALLED"

    fi
}

# ==========================================================
# Status CSS Class
# ==========================================================

status_class() {

    local status="$1"

    case "$status" in

        RUNNING)

            echo "running"

            ;;

        STOPPED)

            echo "stopped"

            ;;

        *)

            echo "unknown"

            ;;

    esac
}
