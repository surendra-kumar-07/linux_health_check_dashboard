# ----------------------------------------------------------
# CPU Usage
# ----------------------------------------------------------

get_cpu_usage() {

    read cpu user nice system idle iowait irq softirq steal guest guest_nice < /proc/stat

    total1=$((user + nice + system + idle + iowait + irq + softirq + steal))
    idle1=$((idle + iowait))

    sleep 1

    read cpu user nice system idle iowait irq softirq steal guest guest_nice < /proc/stat

    total2=$((user + nice + system + idle + iowait + irq + softirq + steal))
    idle2=$((idle + iowait))

    total_diff=$((total2 - total1))
    idle_diff=$((idle2 - idle1))

    if [ "$total_diff" -eq 0 ]; then
        echo "0"
    else
        echo $((100 * (total_diff - idle_diff) / total_diff))
    fi
}

# ----------------------------------------------------------
# Memory Usage
# ----------------------------------------------------------

get_memory_usage() {

    total=$(awk '/MemTotal/ {print $2}' /proc/meminfo)
    available=$(awk '/MemAvailable/ {print $2}' /proc/meminfo)

    used=$((total - available))

    if [ "$total" -gt 0 ]; then
        echo $((used * 100 / total))
    else
        echo "0"
    fi
}

# ----------------------------------------------------------
# Disk Usage
# ----------------------------------------------------------

get_disk_usage() {

    df -P / | awk 'NR==2 {gsub("%","",$5); print $5}'
}

# ----------------------------------------------------------
# Disk Free Space
# ----------------------------------------------------------

get_disk_free() {

    df -h / | awk 'NR==2 {print $4}'
}

# ----------------------------------------------------------
# Disk Total Space
# ----------------------------------------------------------

get_disk_total() {

    df -h / | awk 'NR==2 {print $2}'
}

# ----------------------------------------------------------
# Disk Used Space
# ----------------------------------------------------------

get_disk_used() {

    df -h / | awk 'NR==2 {print $3}'
}

# ----------------------------------------------------------
# Memory Information
# ----------------------------------------------------------

get_memory_total() {

    free -h | awk '/Mem:/ {print $2}'
}

get_memory_used() {

    free -h | awk '/Mem:/ {print $3}'
}

get_memory_available() {

    free -h | awk '/Mem:/ {print $7}'
}

# ----------------------------------------------------------
# Load Average
# ----------------------------------------------------------

get_load_average() {

    awk '{print $1}' /proc/loadavg
}

# ----------------------------------------------------------
# Process Count
# ----------------------------------------------------------

get_process_count() {

    ls -d /proc/[0-9]* 2>/dev/null | wc -l
}

# ----------------------------------------------------------
# CPU Core Count
# ----------------------------------------------------------

get_cpu_cores() {

    nproc
}

# ----------------------------------------------------------
# Uptime
# ----------------------------------------------------------

get_uptime() {

    uptime -p
}

# ----------------------------------------------------------
# Hostname
# ----------------------------------------------------------

get_hostname() {

    hostname
}

# ----------------------------------------------------------
# OS Information
# ----------------------------------------------------------

get_os() {

    if [ -f /etc/os-release ]; then
        . /etc/os-release
        echo "$PRETTY_NAME"
    else
        echo "Unknown Linux"
    fi
}

# ----------------------------------------------------------
# Kernel
# ----------------------------------------------------------

get_kernel() {

    uname -r
}

# ----------------------------------------------------------
# Architecture
# ----------------------------------------------------------

get_architecture() {

    uname -m
}

# ----------------------------------------------------------
# Network Interface
# ----------------------------------------------------------

get_interface() {

    ip route 2>/dev/null |
        awk '/default/ {print $5; exit}'
}

# ----------------------------------------------------------
# Network RX
# ----------------------------------------------------------

get_rx() {

    interface=$(get_interface)

    if [ -n "$interface" ] &&
       [ -f "/sys/class/net/$interface/statistics/rx_bytes" ]; then

        bytes=$(cat "/sys/class/net/$interface/statistics/rx_bytes")

        if [ "$bytes" -ge 1073741824 ]; then
            awk "BEGIN {printf \"%.2f GB\", $bytes/1073741824}"
        elif [ "$bytes" -ge 1048576 ]; then
            awk "BEGIN {printf \"%.2f MB\", $bytes/1048576}"
        elif [ "$bytes" -ge 1024 ]; then
            awk "BEGIN {printf \"%.2f KB\", $bytes/1024}"
        else
            echo "${bytes} B"
        fi
    else
        echo "N/A"
    fi
}

# ----------------------------------------------------------
# Network TX
# ----------------------------------------------------------

get_tx() {

    interface=$(get_interface)

    if [ -n "$interface" ] &&
       [ -f "/sys/class/net/$interface/statistics/tx_bytes" ]; then

        bytes=$(cat "/sys/class/net/$interface/statistics/tx_bytes")

        if [ "$bytes" -ge 1073741824 ]; then
            awk "BEGIN {printf \"%.2f GB\", $bytes/1073741824}"
        elif [ "$bytes" -ge 1048576 ]; then
            awk "BEGIN {printf \"%.2f MB\", $bytes/1048576}"
        elif [ "$bytes" -ge 1024 ]; then
            awk "BEGIN {printf \"%.2f KB\", $bytes/1024}"
        else
            echo "${bytes} B"
        fi
    else
        echo "N/A"
    fi
}

# ----------------------------------------------------------
# Service Status
# ----------------------------------------------------------

service_status() {

    service="$1"

    if systemctl list-unit-files |
        grep -q "^${service}\.service"; then

        if systemctl is-active --quiet "$service"; then
            echo "RUNNING"
        else
            echo "STOPPED"
        fi
    else
        echo "NOT INSTALLED"
    fi
}

# ----------------------------------------------------------
# Status Color
# ----------------------------------------------------------

status_class() {

    status="$1"

    case "$status" in

        RUNNING)
            echo "running"
            ;;

        STOPPED)
            echo "stopped"
            ;;

        *)
            echo "unknown"
            ;;

    esac
}
