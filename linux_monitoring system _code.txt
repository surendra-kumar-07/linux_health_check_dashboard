#!/bin/bash

# ============================================================
# Linux Server Monitoring System
# Version: 1.0
# ============================================================

set -u

# -----------------------------
# Configuration
# -----------------------------

LOG_DIR="$HOME/linux-monitoring/logs"
LOG_FILE="$LOG_DIR/system.log"

CPU_WARNING=70
CPU_CRITICAL=90

MEMORY_WARNING=70
MEMORY_CRITICAL=90

DISK_WARNING=70
DISK_CRITICAL=90

mkdir -p "$LOG_DIR"

# -----------------------------
# Colors
# -----------------------------

RED='\033[0;31m'
YELLOW='\033[1;33m'
GREEN='\033[0;32m'
BLUE='\033[0;34m'
NC='\033[0m'

# -----------------------------
# Get CPU Usage
# -----------------------------

get_cpu_usage() {
    CPU_USAGE=$(top -bn1 | awk '/Cpu\(s\)/ {
        print 100 - $8
    }')

    printf "%.0f" "$CPU_USAGE"
}

# -----------------------------
# Get Memory Usage
# -----------------------------

get_memory_usage() {
    free | awk '/Mem:/ {
        printf "%.0f", ($3 / $2) * 100
    }'
}

# -----------------------------
# Get Disk Usage
# -----------------------------

get_disk_usage() {
    df -P / | awk 'NR==2 {
        gsub("%", "", $5)
        print $5
    }'
}

# -----------------------------
# Get Load Average
# -----------------------------

get_load_average() {
    awk '{print $1, $2, $3}' /proc/loadavg
}

# -----------------------------
# Get Uptime
# -----------------------------

get_uptime() {
    uptime -p 2>/dev/null || uptime
}

# -----------------------------
# Get Process Count
# -----------------------------

get_process_count() {
    ps -e --no-headers | wc -l
}

# -----------------------------
# Get Network RX
# -----------------------------

get_network_rx() {
    awk '
    NR > 2 {
        gsub(":", "", $1)
        rx += $2
    }
    END {
        print rx
    }
    ' /proc/net/dev
}

# -----------------------------
# Get Network TX
# -----------------------------

get_network_tx() {
    awk '
    NR > 2 {
        gsub(":", "", $1)
        tx += $10
    }
    END {
        print tx
    }
    ' /proc/net/dev
}

# -----------------------------
# Convert Bytes to MB
# -----------------------------

bytes_to_mb() {
    BYTES="$1"

    awk -v bytes="$BYTES" 'BEGIN {
        printf "%.2f", bytes / 1024 / 1024
    }'
}

# -----------------------------
# Check Status
# -----------------------------

check_status() {
    VALUE="$1"
    WARNING="$2"
    CRITICAL="$3"

    if [ "$VALUE" -ge "$CRITICAL" ]; then
        echo -e "${RED}CRITICAL${NC}"
    elif [ "$VALUE" -ge "$WARNING" ]; then
        echo -e "${YELLOW}WARNING${NC}"
    else
        echo -e "${GREEN}OK${NC}"
    fi
}

# -----------------------------
# Check Services
# -----------------------------

check_services() {

    echo
    echo "--------------------------------------------"
    echo "SERVICE STATUS"
    echo "--------------------------------------------"

    SERVICES=("sshd" "docker" "nginx")

    for SERVICE in "${SERVICES[@]}"; do

        if systemctl list-unit-files \
            | grep -q "^${SERVICE}.service"; then

            if systemctl is-active --quiet "$SERVICE"; then
                echo -e "$SERVICE : ${GREEN}RUNNING${NC}"
            else
                echo -e "$SERVICE : ${RED}STOPPED${NC}"
            fi

        else
            echo "$SERVICE : NOT INSTALLED"
        fi

    done
}

# -----------------------------
# Check Docker
# -----------------------------

check_docker() {

    echo
    echo "--------------------------------------------"
    echo "DOCKER STATUS"
    echo "--------------------------------------------"

    if command -v docker >/dev/null 2>&1; then

        if systemctl is-active --quiet docker 2>/dev/null; then

            CONTAINER_COUNT=$(docker ps -q 2>/dev/null | wc -l)

            echo -e "Docker       : ${GREEN}RUNNING${NC}"
            echo "Containers   : $CONTAINER_COUNT"

            if [ "$CONTAINER_COUNT" -gt 0 ]; then
                echo
                docker ps --format \
                "table {{.Names}}\t{{.Status}}\t{{.Ports}}"
            fi

        else
            echo -e "Docker       : ${RED}STOPPED${NC}"
        fi

    else
        echo "Docker       : NOT INSTALLED"
    fi
}

# -----------------------------
# Top CPU Processes
# -----------------------------

top_cpu_processes() {

    echo
    echo "--------------------------------------------"
    echo "TOP CPU PROCESSES"
    echo "--------------------------------------------"

    ps -eo pid,user,comm,%cpu,%mem --sort=-%cpu \
        | head -6
}

# -----------------------------
# Top Memory Processes
# -----------------------------

top_memory_processes() {

    echo
    echo "--------------------------------------------"
    echo "TOP MEMORY PROCESSES"
    echo "--------------------------------------------"

    ps -eo pid,user,comm,%cpu,%mem --sort=-%mem \
        | head -6
}

# -----------------------------
# Main Monitoring Function
# -----------------------------

monitor_system() {

    TIMESTAMP=$(date '+%Y-%m-%d %H:%M:%S')
    HOSTNAME=$(hostname)

    CPU=$(get_cpu_usage)
    MEMORY=$(get_memory_usage)
    DISK=$(get_disk_usage)
    LOAD=$(get_load_average)
    UPTIME=$(get_uptime)
    PROCESSES=$(get_process_count)

    RX=$(get_network_rx)
    TX=$(get_network_tx)

    RX_MB=$(bytes_to_mb "$RX")
    TX_MB=$(bytes_to_mb "$TX")

    CPU_STATUS=$(check_status \
        "$CPU" \
        "$CPU_WARNING" \
        "$CPU_CRITICAL")

    MEMORY_STATUS=$(check_status \
        "$MEMORY" \
        "$MEMORY_WARNING" \
        "$MEMORY_CRITICAL")

    DISK_STATUS=$(check_status \
        "$DISK" \
        "$DISK_WARNING" \
        "$DISK_CRITICAL")

    clear

    echo "============================================================"
    echo "              LINUX SERVER MONITORING SYSTEM"
    echo "============================================================"

    echo
    echo "Hostname       : $HOSTNAME"
    echo "Date & Time    : $TIMESTAMP"
    echo "Uptime         : $UPTIME"

    echo
    echo "------------------------------------------------------------"
    echo "SYSTEM RESOURCES"
    echo "------------------------------------------------------------"

    printf "CPU Usage      : %s%%       " "$CPU"
    echo -e "$CPU_STATUS"

    printf "Memory Usage   : %s%%       " "$MEMORY"
    echo -e "$MEMORY_STATUS"

    printf "Disk Usage     : %s%%       " "$DISK"
    echo -e "$DISK_STATUS"

    echo "Load Average   : $LOAD"
    echo "Processes      : $PROCESSES"

    echo
    echo "------------------------------------------------------------"
    echo "NETWORK"
    echo "------------------------------------------------------------"

    echo "Network RX     : ${RX_MB} MB"
    echo "Network TX     : ${TX_MB} MB"

    check_services

    check_docker

    top_cpu_processes

    top_memory_processes

    echo
    echo "------------------------------------------------------------"
    echo "LOG"
    echo "------------------------------------------------------------"

    echo "Log file       : $LOG_FILE"

    echo
    echo "============================================================"

    # -----------------------------
    # Write monitoring data to log
    # -----------------------------

    {
        echo "============================================================"
        echo "Timestamp      : $TIMESTAMP"
        echo "Hostname       : $HOSTNAME"
        echo "CPU Usage      : ${CPU}%"
        echo "Memory Usage   : ${MEMORY}%"
        echo "Disk Usage     : ${DISK}%"
        echo "Load Average   : $LOAD"
        echo "Uptime         : $UPTIME"
        echo "Processes      : $PROCESSES"
        echo "Network RX     : ${RX_MB} MB"
        echo "Network TX     : ${TX_MB} MB"
        echo "============================================================"
        echo
    } >> "$LOG_FILE"
}

# -----------------------------
# Menu
# -----------------------------

show_menu() {

    while true; do

        echo
        echo "============================================================"
        echo "              LINUX MONITORING SYSTEM"
        echo "============================================================"
        echo
        echo "1. Run monitoring once"
        echo "2. Continuous monitoring"
        echo "3. Show logs"
        echo "4. Show system information"
        echo "5. Exit"
        echo

        read -r -p "Enter your choice: " CHOICE

        case "$CHOICE" in

            1)
                monitor_system
                ;;

            2)
                read -r -p "Enter interval in seconds [10]: " INTERVAL

                if [ -z "$INTERVAL" ]; then
                    INTERVAL=10
                fi

                if ! [[ "$INTERVAL" =~ ^[0-9]+$ ]]; then
                    echo "Invalid interval."
                    continue
                fi

                while true; do
                    monitor_system
                    echo
                    echo "Next check in $INTERVAL seconds..."
                    echo "Press Ctrl+C to stop."

                    sleep "$INTERVAL"
                done
                ;;

            3)
                if [ -f "$LOG_FILE" ]; then
                    cat "$LOG_FILE"
                else
                    echo "No logs available."
                fi
                ;;

            4)
                echo
                echo "================ SYSTEM INFORMATION ================"
                echo
                echo "Hostname:"
                hostname

                echo
                echo "Operating System:"
                if [ -f /etc/os-release ]; then
                    . /etc/os-release
                    echo "$PRETTY_NAME"
                else
                    uname -s
                fi

                echo
                echo "Kernel:"
                uname -r

                echo
                echo "Architecture:"
                uname -m

                echo
                echo "IP Address:"
                hostname -I 2>/dev/null || true

                echo
                echo "===================================================="
                ;;

            5)
                echo
                echo "Exiting Linux Monitoring System."
                exit 0
                ;;

            *)
                echo
                echo "Invalid choice. Please select 1-5."
                ;;

        esac

    done
}

# -----------------------------
# Start Application
# -----------------------------

show_menu
